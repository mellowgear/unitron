|static inline uint32_t shift_op(int kind, uint32_t v, int n, int bits) {
|    uint32_t m = mask_of(bits), top = 1u << (bits - 1);
|    v &= m;

|        case 1: out = v & 1; v = (v >> 1) | (out ? top : 0); break;                     /* RRC */

|        default: out = v & 1; v >>= 1; break;                                           /* SRL */
|        }
