# Golden CLI cases. Each case: files (written into an empty directory), then commands run there.
# tools/freeze_vectors.py golden runs them with the Python reference and writes expected.txt;
# the C++ test runner (tests/cpp/golden.cpp) runs them with the unitron binary and compares.
# Format: "=== name", "--- file" (text, ends at the next marker), "--- file hex" (hex bytes),
# "$ args" (one unitron command). Lines starting with "#" outside files are comments.

--- tiny.asm
$maximum
	module	tiny
h	section	romdata abs=0x200000
	db	" LICENSED BY SNK CORPORATION"
	dl	start
	dw	0
	db	0,0x10
	db	"TINY        "
	dfb	16,0
c	section	code abs=0x200100
start:
	di
	ld	XSP,0x6C00
	ld	XHL,irq
	ld	(0x6FCC),XHL
	ld	XWA,timer
	ld	(0x6FD4),XWA
	ld	A,(0x6F82)
	ld	(0x62AA:24),A
	ld	BC,2:16
	ld	C,(XHL+0xF:16)
	cp	E,0:8
	ld	WA,BC
	mul	WA,5
	cp	XHL,0
	ld	(XIX+),DE
	ld	WA,(-XHL:1)
	lda	XHL,XHL+DE
	calr	subr
	call	subr
	jp	C,XIX+2
	ld	RW3,0x12
	swi	1
	swi	1
	ld	(0x8002),A
	ldar	XIY,table
	ld	A,(XHL+A)
	ld	(0x8000),A
	ld	(0x87E2),A
	bit	3,(0x8010)
loop:	djnz	B,loop
	jr	loop
subr:	push	WA
	pop	WA
	ret
irq:	reti
timer:	ld	A,1
	ret	Z
	jr	timer
table:	db	1,2,3
	dl	hidden
	db	0xFF,0xFF
hidden:	ld	A,2
	call	hidden2
	ret
	db	0x55,0x66
hidden2:	ld	B,3
	ret
	db	0x50,0x51
	db	0x1F,0xFF,0xFF,0xFF
	db	0,0,0,0
	end

=== lcf
--- main.asm
$maximum
	module	main
	extern	large	__DataRAM
	public	start
	extern	medium	counter
vec	section	code abs=0x200000
	jp	start
	dl	__DataRAM
f_code	section	code
start:	ld	XWA,init
	ld	(counter),XWA
	ret
f_const	section	romdata
init:	dw	0x1234
f_data	section	data
iv:	dw	5
	end
--- sub.asm
$maximum
	module	sub
	public	counter
f_area	section	data medium
counter	dsw	1
f_code	section	code
	nop
	end
--- cart.lcf
/* a cartridge layout */
memory
{
    ram(RW) : org=0x4000, len=0x1000
    rom(RX) : org=0x200000, len=0x1000
}
sections
{
    ram_data addr=0x4000 : {*(f_area)}
    code addr=0x200010 : {*(f_code)}
    rodata addr=addr(code)+sizeof(code) : {*(f_const)}
    roinit org=org(rodata)+sizeof(rodata) addr=addr(ram_data)+sizeof(ram_data) : {*(f_data)}
    __DataRAM = addr(roinit);
}
--- bad.lcf
sections { x addr=nosuch( : {*(f_code)} }
$ asm main.asm sub.asm --lcf cart.lcf -o cart.bin --origin 0x200000 --sym cart.sym
$ asm main.asm sub.asm --lcf bad.lcf -o bad.bin
$ asm main.asm sub.asm --lcf cart.lcf --place f_code=0x200100 -o x.bin
